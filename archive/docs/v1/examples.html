1. Form + Table Combination (samples/v1/form-and-table)

Launch Demo

Renders an input form on top of a data table from shared columns and records.

import render from "json-renderers-from-instructions-collection";
import columns from "./columns.json" with { type: "json" };
import data from "./data.json" with { type: "json" };

render({
    data,
    columns,
    targetHtmlId: "dom-render-container",
    appendPosition: "replace",
    renderers: [
        { type: "form.simple", options: {} },
        { 
            type: "table.simple", 
            options: { showSerial: true, showSearch: true, showOptions: true } 
        }
    ]
});

3. Interactive Event Loop (samples/interactive/v1)

Launch Interactive POC

Implements live real-time search filtering, form submit adding items to dataset, and row-level button action dispatching:

import render from "json-renderers-from-instructions-collection";
import columns from "./columns.json" with { type: "json" };
import initialData from "./data.json" with { type: "json" };

let dataset = [...initialData];

const renderApp = () => {
    // 1. Mount Collection
    render({
        data: dataset,
        columns,
        targetHtmlId: "dom-render-container",
        renderers: [
            { type: "form.simple", options: {} },
            { 
                type: "table.simple", 
                options: { showSerial: true, showSearch: true, showOptions: true } 
            }
        ]
    });

    // 2. Attach Live Search Filter
    const searchInput = document.getElementById("tableSearch");
    if (searchInput) {
        searchInput.addEventListener("input", (e) => {
            const query = e.target.value.toLowerCase();
            document.querySelectorAll('[data-role="table-body"] tr').forEach(row => {
                row.style.display = row.textContent.toLowerCase().includes(query) ? "" : "none";
            });
        });
    }

    // 3. Form Submit Loop
    const form = document.querySelector('form[data-role="form-body"]');
    if (form) {
        form.addEventListener("submit", (e) => {
            e.preventDefault();
            const formData = new FormData(form);
            dataset.push({
                itemName: formData.get("itemName"),
                category: formData.get("category"),
                baseUnit: formData.get("baseUnit")
            });
            renderApp(); // Re-render updates table & serial numbers
        });
    }

    // 4. Row Action Delegation
    const tbody = document.querySelector('[data-role="table-body"]');
    if (tbody) {
        tbody.addEventListener("click", (e) => {
            const actionBtn = e.target.closest(".ks-btn-show");
            if (!actionBtn) return;
            const tr = actionBtn.closest("tr");
            console.log("Clicked row:", tr.textContent);
        });
    }
};

renderApp();
