3. Interactive Loop: Live Search + Add Form (samples/interactive/v1)

Launch Interactive POC

Shows how the rendered DOM cleanly supports real-time live filtering, form submits with dataset mutations, and row action clicks:

let dataset = [...initialData];

const renderApp = () => {
    render({
        data: dataset,
        columns,
        targetHtmlId: "dom-render-container",
        renderers: [
            { type: "form.simple" },
            { 
                type: "table.simple", 
                options: { showSerial: true, showSearch: true, showOptions: true } 
            }
        ]
    });

    // 1. Live Instant Filter
    const searchInput = document.getElementById("tableSearch");
    if (searchInput) {
        searchInput.addEventListener("input", (e) => {
            const query = e.target.value.toLowerCase();
            document.querySelectorAll('[data-role="table-body"] tr').forEach(row => {
                row.style.display = row.textContent.toLowerCase().includes(query) ? "" : "none";
            });
        });
    }

    // 2. Form Add Loop
    const form = document.querySelector('form[data-role="form-body"]');
    if (form) {
        form.addEventListener("submit", (e) => {
            e.preventDefault();
            const formData = new FormData(form);
            dataset.push({
                itemName: formData.get("itemName"),
                category: formData.get("category"),
                baseUnit: formData.get("baseUnit")
            });
            renderApp(); // Re-render updates table & serial numbers seamlessly
        });
    }
};

renderApp();
